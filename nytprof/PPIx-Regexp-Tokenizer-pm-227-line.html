575 or return;

577 # the returned object until I am done with all its children.
578 my $ppi;

580 $ppi = $token->ppi();

582 $ppi->find( 'PPI::Token::Operator' ) || [] };
583 $iterator = sub {
584 my $op = shift @ops
585 or return;
586 return $op->snext_sibling();
587 };
588 } elsif ( $iterator->isa( 'PPI::Element' ) ) {
589 my @eles = ( $iterator );
590 $iterator = sub {
591 return shift @eles;
592 };
593 } elsif ( CODE_REF ne ref $iterator ) {

610 # they should be parsed as casts, but ...
